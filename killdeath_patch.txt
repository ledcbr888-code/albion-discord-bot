async function fetchKillEventDetails(eventId) {
    const id = String(eventId || '').trim();
    if (!id) return null;
    const urls = [`https://gameinfo-sgp.albiononline.com/api/gameinfo/events/${encodeURIComponent(id)}`, `https://gameinfo.albiononline.com/api/gameinfo/events/${encodeURIComponent(id)}`];
    for (const url of urls) {
        try {
            const response = await axios.get(url, { timeout: 15000, headers: { 'User-Agent': 'Albion-Discord-Bot/1.0', Accept: 'application/json' } });
            if (response.data && typeof response.data === 'object') return response.data;
        } catch (_) {}
    }
    return null;
}

function getEquipmentSlots(player) {
    const eq = player?.Equipment || player?.equipment || {};
    const slotNames = ['MainHand', 'OffHand', 'Head', 'Armor', 'Shoes', 'Bag', 'Cape', 'Mount', 'Food', 'Potion'];
    return slotNames.map(slot => {
        const item = eq[slot] ?? eq[slot.toLowerCase()] ?? null;
        if (!item || typeof item !== 'object') return { slot, item: null };
        return { slot, item, type: normalizeAlbionItemId(item.Type ?? item.type ?? item.ItemId ?? item.itemId ?? ''), quality: Number(item.Quality ?? item.quality ?? 0) || 0, count: Number(item.Count ?? item.count ?? 1) || 1 };
    });
}
function getPlayerItemPower(player) { return Number(player?.AverageItemPower ?? player?.averageItemPower ?? player?.IP ?? player?.ip ?? 0) || 0; }
function getPlayerDamage(player) { return Number(player?.DamageDone ?? player?.damageDone ?? player?.Damage ?? player?.damage ?? 0) || 0; }
function getPlayerHealing(player) { return Number(player?.SupportHealingDone ?? player?.supportHealingDone ?? player?.Healing ?? player?.healing ?? 0) || 0; }

async function fetchMarketPrices(entries) {
    const ids = [...new Set((entries || []).filter(e => e?.type).map(e => normalizeAlbionItemId(e.type)).filter(Boolean))];
    if (!ids.length) return new Map();
    try {
        const url = `https://east.albion-online-data.com/api/v2/stats/prices/${ids.map(encodeURIComponent).join(',')}.json?locations=Caerleon,Bridgewatch,Thetford,Fort%20Sterling,Lymhurst,Martlock,Black%20Market`;
        const response = await axios.get(url, { timeout: 12000, headers: { 'User-Agent': 'Albion-Discord-Bot/1.0', Accept: 'application/json' } });
        const map = new Map();
        for (const row of (Array.isArray(response.data) ? response.data : [])) {
            const id = normalizeAlbionItemId(row.item_id || row.itemId || '');
            const q = Number(row.quality || 0) || 0;
            const price = Number(row.sell_price_min || 0) || 0;
            if (!id || price <= 0) continue;
            const key = `${id}|${q}`;
            if (!map.get(key) || price < map.get(key)) map.set(key, price);
            if (!map.get(`${id}|0`) || price < map.get(`${id}|0`)) map.set(`${id}|0`, price);
        }
        return map;
    } catch (err) { console.warn('⚠️ Market price lookup skipped:', err.message); return new Map(); }
}
function getEntryPrice(entry, priceMap) { if (!entry?.type) return 0; const id = normalizeAlbionItemId(entry.type), q = Number(entry.quality || 0) || 0; return Number(priceMap.get(`${id}|${q}`) || priceMap.get(`${id}|0`) || 0) || 0; }

async function loadAlbionItemImage(type, quality = 0, size = 180) {
    const id = normalizeAlbionItemId(type); if (!id) return null; const cleanId = id.replace(/\.png$/i, '');
    const urls = [`https://render.albiononline.com/v1/item/${encodeURIComponent(cleanId)}.png?quality=${Math.max(0, Math.min(5, Number(quality) || 0))}&size=${size}`, `https://render.albiononline.com/v1/item/${encodeURIComponent(cleanId)}.png?size=${size}`, `https://render.albiononline.com/v1/item/${encodeURIComponent(cleanId.split('@')[0])}.png?size=${size}`];
    for (const url of urls) { try { return await Promise.race([loadImage(url), new Promise((_, reject) => setTimeout(() => reject(new Error('image timeout')), 5000))]); } catch (_) {} }
    return null;
}
async function loadAlbionAvatarImage(player, size = 120) {
    const avatar = player?.Avatar || player?.avatar; if (!avatar) return null;
    const urls = [`https://render.albiononline.com/v1/avatar/${encodeURIComponent(avatar)}.png?size=${size}`, `https://render.albiononline.com/v1/avatar/${encodeURIComponent(avatar)}.png`];
    for (const url of urls) { try { return await Promise.race([loadImage(url), new Promise((_, reject) => setTimeout(() => reject(new Error('avatar timeout')), 4000))]); } catch (_) {} }
    return null;
}
function drawTextFit(ctx, text, x, y, maxWidth, fontSize, weight = 'bold') { let size = fontSize; const value = String(text ?? ''); while (size > 12) { ctx.font = `${weight} ${size}px Arial`; if (ctx.measureText(value).width <= maxWidth) break; size--; } ctx.fillText(value, x, y); }
function roundedRect(ctx, x, y, w, h, r, fill, stroke = null, lineWidth = 1) { ctx.beginPath(); ctx.roundRect(x, y, w, h, r); if (fill) { ctx.fillStyle = fill; ctx.fill(); } if (stroke) { ctx.lineWidth = lineWidth; ctx.strokeStyle = stroke; ctx.stroke(); } }
async function drawItemSlot(ctx, x, y, size, entry, labelValue = '') {
    roundedRect(ctx, x, y, size, size, 10, '#25272a', '#3e4246', 2); if (!entry?.type) return;
    const img = await loadAlbionItemImage(entry.type, entry.quality, size); if (img) ctx.drawImage(img, x, y, size, size);
    if (entry.count > 1) { ctx.fillStyle = '#fff'; ctx.font = 'bold 18px Arial'; ctx.textAlign = 'right'; ctx.fillText(String(entry.count), x + size - 7, y + size - 7); ctx.textAlign = 'left'; }
    if (labelValue) { ctx.font = 'bold 15px Arial'; const w = Math.min(size - 8, Math.max(50, ctx.measureText(labelValue).width + 14)); roundedRect(ctx, x + 4, y + 4, w, 26, 8, '#19e84a'); ctx.fillStyle = '#061007'; ctx.fillText(labelValue, x + 11, y + 22); }
}
async function drawPlayerHeader(ctx, player, x, y, width, align = 'left') {
    const name = player?.Name || player?.name || 'Unknown', guild = player?.GuildName || player?.guildName || player?.guild || 'No Guild', avatarSize = 82;
    const avatarX = align === 'left' ? x : x + width - avatarSize, textX = align === 'left' ? x + avatarSize + 14 : x, maxText = width - avatarSize - 18;
    roundedRect(ctx, avatarX, y, avatarSize, avatarSize, 50, '#1d2023', '#70757a', 2); const avatar = await loadAlbionAvatarImage(player, avatarSize);
    if (avatar) { ctx.save(); ctx.beginPath(); ctx.arc(avatarX + avatarSize / 2, y + avatarSize / 2, avatarSize / 2 - 3, 0, Math.PI * 2); ctx.clip(); ctx.drawImage(avatar, avatarX + 3, y + 3, avatarSize - 6, avatarSize - 6); ctx.restore(); }
    else { ctx.fillStyle = '#3a3f44'; ctx.beginPath(); ctx.arc(avatarX + avatarSize / 2, y + avatarSize / 2, avatarSize / 2 - 4, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = '#fff'; ctx.font = 'bold 30px Arial'; ctx.textAlign = 'center'; ctx.fillText(name.charAt(0).toUpperCase(), avatarX + avatarSize / 2, y + 52); ctx.textAlign = 'left'; }
    ctx.fillStyle = '#fff'; drawTextFit(ctx, name, textX, y + 35, maxText, 28); ctx.fillStyle = '#ddd'; drawTextFit(ctx, guild, textX, y + 65, maxText, 18, 'normal');
}
function mergePlayersForEvent(detail) {
    const list = []; const push = p => { if (!p || !p.Name && !p.name) return; const name = String(p.Name ?? p.name), key = name.toLowerCase(); const existing = list.find(x => String(x.Name ?? x.name).toLowerCase() === key); if (!existing) list.push({ ...p }); else { if (!existing.Equipment && p.Equipment) existing.Equipment = p.Equipment; if (!existing.Avatar && p.Avatar) existing.Avatar = p.Avatar; if (!existing.GuildName && p.GuildName) existing.GuildName = p.GuildName; if (!existing.AverageItemPower && p.AverageItemPower) existing.AverageItemPower = p.AverageItemPower; existing.DamageDone = Math.max(Number(existing.DamageDone || 0), Number(p.DamageDone || 0)); existing.SupportHealingDone = Math.max(Number(existing.SupportHealingDone || 0), Number(p.SupportHealingDone || 0)); } };
    (Array.isArray(detail?.Participants) ? detail.Participants : []).forEach(push); (Array.isArray(detail?.GroupMembers) ? detail.GroupMembers : []).forEach(push); push(detail?.Killer); push(detail?.Victim); return list;
}

async function buildKillDeathImages(eventData, isDeath = false) {
    let detail = eventData?.detail || null; if (!detail && eventData?.eventId) detail = await fetchKillEventDetails(eventData.eventId);
    const killer = detail?.Killer || eventData?.killer || { Name: eventData?.killerName, GuildName: eventData?.killerGuild }, victim = detail?.Victim || eventData?.victim || { Name: eventData?.victimName, GuildName: eventData?.victimGuild };
    const totalFame = Number(detail?.TotalVictimKillFame ?? eventData?.totalFame ?? 0) || 0, battleId = detail?.BattleId || detail?.battleId || eventData?.battleId || '', eventId = detail?.EventId || eventData?.eventId || '', eventTime = detail?.TimeStamp || eventData?.eventTime || Date.now();

    const mainW = 1080, mainH = 752, canvas = createCanvas(mainW, mainH), ctx = canvas.getContext('2d'); ctx.fillStyle = '#2d3033'; ctx.fillRect(0, 0, mainW, mainH);
    const leftX = 34, rightX = 674, headerY = 22; await drawPlayerHeader(ctx, killer, leftX, headerY, 400, 'left'); await drawPlayerHeader(ctx, victim, rightX, headerY, 372, 'right');
    ctx.fillStyle = '#eee'; ctx.font = 'bold 15px Arial'; ctx.textAlign = 'center'; ctx.fillText('The hottest killboard in your neighbourhood', mainW / 2, 102); ctx.font = 'bold 23px Arial'; ctx.fillText('www.albionkb.com', mainW / 2, 127); ctx.textAlign = 'left';

    const leftEq = getEquipmentSlots(killer), rightEq = getEquipmentSlots(victim), useful = ['Head', 'Armor', 'Shoes', 'MainHand', 'OffHand', 'Food', 'Potion', 'Mount'];
    const leftEntries = useful.map(s => leftEq.find(e => e.slot === s)).filter(Boolean), rightEntries = useful.map(s => rightEq.find(e => e.slot === s)).filter(Boolean);
    const victimInventory = Array.isArray(victim?.Inventory) ? victim.Inventory.map(item => ({ slot: 'Inventory', item, type: normalizeAlbionItemId(item?.Type ?? item?.type ?? ''), quality: Number(item?.Quality ?? item?.quality ?? 0) || 0, count: Number(item?.Count ?? item?.count ?? 1) || 1 })).filter(e => e.type).slice(0, 8) : [];
    const priceMap = await fetchMarketPrices([...leftEntries, ...rightEntries, ...victimInventory]);
    const gridY = 150, slot = 102, gap = 10, positions = [[0,0],[1,0],[2,0],[0,1],[1,1],[2,1],[0,2],[1,2],[1,3]];
    for (let i = 0; i < leftEntries.length && i < positions.length; i++) { const [cx, cy] = positions[i], price = getEntryPrice(leftEntries[i], priceMap); await drawItemSlot(ctx, leftX + cx * (slot + gap), gridY + cy * (slot + gap), slot, leftEntries[i], price > 0 ? formatFame(price) : ''); }
    for (let i = 0; i < rightEntries.length && i < positions.length; i++) { const [cx, cy] = positions[i], price = getEntryPrice(rightEntries[i], priceMap); await drawItemSlot(ctx, rightX + cx * (slot + gap), gridY + cy * (slot + gap), slot, rightEntries[i], price > 0 ? formatFame(price) : ''); }

    ctx.textAlign = 'center'; ctx.fillStyle = '#fff'; ctx.font = 'bold 30px Arial'; ctx.fillText(isDeath ? 'DEATH' : 'KILL', mainW / 2, 195); ctx.font = 'bold 32px Arial'; ctx.fillText(Number(totalFame).toLocaleString(), mainW / 2, 255); ctx.font = '20px Arial'; ctx.fillText('Kill Fame', mainW / 2, 282);
    const leftIP = getPlayerItemPower(killer), rightIP = getPlayerItemPower(victim); ctx.font = 'bold 18px Arial'; ctx.fillText(`IP ${leftIP ? leftIP.toFixed(0) : 'N/A'}`, 225, 505); ctx.fillText(`IP ${rightIP ? rightIP.toFixed(0) : 'N/A'}`, 855, 505);
    ctx.font = '20px Arial'; ctx.fillStyle = '#e8e8e8'; const utcText = new Intl.DateTimeFormat('en-GB', { timeZone: 'UTC', day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit', hour12: false }).format(new Date(eventTime)).replace(',', ''); ctx.fillText(`${utcText} UTC`, mainW / 2, 545);
    const leftEqValue = leftEntries.reduce((sum, e) => sum + getEntryPrice(e, priceMap) * Math.max(1, e.count || 1), 0), rightEqValue = rightEntries.reduce((sum, e) => sum + getEntryPrice(e, priceMap) * Math.max(1, e.count || 1), 0); ctx.font = 'bold 19px Arial'; ctx.fillStyle = '#f4f4f4'; ctx.fillText(`EQ: ${leftEqValue ? formatFame(leftEqValue) : `${leftEq.filter(e => e.item).length} items`}`, 225, 585); ctx.fillText(`EQ: ${rightEqValue ? formatFame(rightEqValue) : `${rightEq.filter(e => e.item).length} items`}`, 855, 585);
    for (let i = 0; i < victimInventory.length; i++) { const x = 38 + i * 118, price = getEntryPrice(victimInventory[i], priceMap); await drawItemSlot(ctx, x, 635, 82, victimInventory[i], price > 0 ? formatFame(price) : ''); }
    ctx.fillStyle = '#8b8f93'; ctx.fillRect(0, 620, mainW, 2); ctx.textAlign = 'left'; ctx.fillStyle = '#d8d8d8'; ctx.font = '16px Arial'; ctx.fillText(`Battle ID: ${battleId || 'N/A'}`, 34, 650); ctx.fillText(`Event ID: ${eventId || 'N/A'}`, 34, 676); ctx.fillText(`Participants: ${Array.isArray(detail?.Participants) ? detail.Participants.length : 0}`, 34, 702); ctx.textAlign = 'right'; ctx.fillText(isDeath ? 'Death Tracker' : 'Kill Tracker', mainW - 34, 650); ctx.fillText('Albion Online Killboard Style', mainW - 34, 676); ctx.textAlign = 'left';
    const main = new AttachmentBuilder(canvas.toBuffer('image/png'), { name: isDeath ? 'death-report.png' : 'kill-report.png' });

    const participants = mergePlayersForEvent(detail).filter(p => { const damage = getPlayerDamage(p), heal = getPlayerHealing(p); return damage > 0 || heal > 0 || String(p.Name || '').toLowerCase() === String(killer.Name || '').toLowerCase(); });
    const totalDamage = participants.reduce((sum, p) => sum + getPlayerDamage(p), 0) || 1, detailW = 1280, rowH = 220, detailH = Math.max(210, 90 + participants.length * rowH), detailCanvas = createCanvas(detailW, detailH), dctx = detailCanvas.getContext('2d'); dctx.fillStyle = '#2d3033'; dctx.fillRect(0, 0, detailW, detailH);
    const killerDamage = getPlayerDamage(killer), top = killerDamage > 0 ? killerDamage : Math.max(...participants.map(getPlayerDamage), 0), yellow = Math.max(0, Math.min(100, ((top || 0) / totalDamage) * 100));
    roundedRect(dctx, 40, 24, detailW - 80, 58, 28, '#ef476f', '#111111', 3); roundedRect(dctx, 40, 24, (detailW - 80) * (yellow / 100), 58, 28, '#ffb52e', '#111111', 3); dctx.fillStyle = '#080808'; dctx.font = 'bold 24px Arial'; dctx.textAlign = 'center'; dctx.fillText(`${yellow.toFixed(1)}%`, 40 + (detailW - 80) * (yellow / 100) / 2, 61); dctx.fillText(`${(100 - yellow).toFixed(1)}%`, 40 + (detailW - 80) * (0.5 + yellow / 200), 61); dctx.textAlign = 'left';
    const ordered = participants.sort((a, b) => getPlayerDamage(b) - getPlayerDamage(a));
    for (let i = 0; i < ordered.length; i++) { const p = ordered[i], y = 105 + i * rowH; roundedRect(dctx, 50, y, detailW - 100, rowH - 28, 24, '#383c40'); const dmg = getPlayerDamage(p), pct = totalDamage > 0 ? dmg / totalDamage * 100 : 0, weapon = getEquipmentSlots(p).find(e => e.slot === 'MainHand'); if (weapon) await drawItemSlot(dctx, 76, y + 42, 125, weapon); await drawPlayerHeader(dctx, p, 230, y + 22, 410, 'left'); dctx.fillStyle = '#fff'; dctx.font = 'bold 18px Arial'; dctx.fillText(`IP ${getPlayerItemPower(p) ? getPlayerItemPower(p).toFixed(0) : 'N/A'}`, 660, y + 40); dctx.fillStyle = '#ffb52e'; dctx.font = 'bold 22px Arial'; dctx.fillText(`${Math.round(dmg).toLocaleString()} dmg`, 660, y + 72); dctx.fillStyle = '#f0f0f0'; dctx.font = '18px Arial'; dctx.fillText(`dealt ${pct.toFixed(1)}% of total damage`, 660, y + 103); const slots = getEquipmentSlots(p).filter(e => ['MainHand','Head','Armor','Shoes','OffHand','Cape','Food','Potion'].includes(e.slot)); for (let j = 0; j < slots.length && j < 8; j++) await drawItemSlot(dctx, 230 + j * 108, y + 118, 88, slots[j]); }
    dctx.fillStyle = '#fff'; dctx.font = 'bold 18px Arial'; dctx.textAlign = 'right'; dctx.fillText(`Event ${eventId || 'N/A'}`, detailW - 50, detailH - 18); dctx.textAlign = 'left';
    const details = new AttachmentBuilder(detailCanvas.toBuffer('image/png'), { name: isDeath ? 'death-details.png' : 'kill-details.png' }); return { main, details };
}
async function generateKillDeathCardImage(eventData, isDeath = false) { const images = await buildKillDeathImages(eventData, isDeath); return images.main; }
